CREATE TABLE movie (
    movie_id INTEGER PRIMARY KEY,
    title VARCHAR(100) NOT NULL,
    genre VARCHAR(50),
    release_year INTEGER,
    rating DECIMAL(3,1),
    price DECIMAL(10,2)
);

CREATE TABLE customer (
    customer_id INTEGER PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    city VARCHAR(50),
    age INTEGER,
    subscription VARCHAR(30)
);

CREATE TABLE watch_history (
    watch_id INTEGER PRIMARY KEY,
    customer_id INTEGER,
    movie_id INTEGER,
    hours_watched DECIMAL(5,2),
    FOREIGN KEY (customer_id) REFERENCES customer(customer_id),
    FOREIGN KEY (movie_id) REFERENCES movie(movie_id)
);

then determine, who movie watched hours?
which movies was watched the most?
calculate the total hours watched for each movie?

1. Who watched which movie and for how many hours?
SELECT
    c.customer_name AS "Customer Name",
    m.title AS "Movie",
    w.hours_watched AS "Hours Watched"
FROM customer AS c
JOIN watch_history AS w
    ON c.customer_id = w.customer_id
JOIN movie AS m
    ON w.movie_id = m.movie_id;

2. Which movie was watched the most?

Use SUM() and sort from highest to lowest:

SELECT
    m.title AS "Movie",
    SUM(w.hours_watched) AS "Total Hours"
FROM movie AS m
JOIN watch_history AS w
    ON m.movie_id = w.movie_id
GROUP BY m.movie_id, m.title
ORDER BY "Total Hours" DESC
LIMIT 1;

3. Calculate total hours watched for each movie
SELECT
    m.title AS "Movie",
    SUM(w.hours_watched) AS "Total Hours"
FROM movie AS m
JOIN watch_history AS w
    ON m.movie_id = w.movie_id
GROUP BY m.movie_id, m.title
ORDER BY "Total Hours" DESC;
